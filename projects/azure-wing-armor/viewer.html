    Phaser 3

    // preload
this.load.atlas('azure', 'atlas/azure_wing@2x.png', 'atlas/azure_wing@2x.json');
this.load.json('azureSet', 'azure_wing_set.json');

// create: ráp bộ giáp quanh gốc nhân vật (x, y = giữa hai bàn chân)
const set = this.cache.json.get('azureSet'), k = 2; // atlas @2x
const view = 'front';                               // 'back' khi nhân vật quay lưng
Object.entries(set.pieces)
  .sort((a, b) => a[1].z - b[1].z)
  .forEach(([name, piece]) => {
    const p = piece.views[view];
    const [ox, oy] = p.offset_1x, [px, py] = p.pivot, [w, h] = p.size_1x;
    this.add.image(x + (ox + px * w) * k * s, y + (oy + py * h) * k * s, 'azure', `${view}/${name}`)
      .setOrigin(px, py).setScale(s);   // s = tỉ lệ nhân vật
  });
